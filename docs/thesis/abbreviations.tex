\subsection*{Mathematical symbols}

\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\textbf{Symbol} & \textbf{Meaning} \\
\midrule
$I(x,y;t)$         & image intensity at pixel $(x,y)$ and diffusion iteration $t$ \\
$q(x,y;t)$         & instantaneous coefficient of variation \\
$T$                & temperature scaling parameter (calibration) \\
$N$                & number of MC-Dropout forward passes (here 50) \\
$w_c$              & per-class loss weight, $w_c = n_{\text{total}}/(2 n_c)$ \\
$\sigma$           & logistic sigmoid; here also Gaussian blur standard deviation \\
$\mathbb{H}$       & predictive entropy (nats) \\
$\epsilon$         & small numerical-stability constant \\
$\theta$           & model parameters \\
$\hat{y}$          & predicted label; $p$ predicted probability of \PCOS{} class \\
\bottomrule
\end{tabularx}
